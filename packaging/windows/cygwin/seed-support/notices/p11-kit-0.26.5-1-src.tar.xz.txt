src_install() {
	cd $S					# 0.25.7
	meson_install				# 0.25.7

	cd $B
#	cyginstall				0.25.7
	lndirs $S/_build.$ARCH-pc-cygwin $B	# 0.25.7

	dosym ../bin/cyg$NAME-0.dll /usr/lib/$NAME-proxy.so

	mv $D/etc/$CRYPTOKI/$CRYPTOKI.conf{.example,}
	make_etc_defaults /etc/$CRYPTOKI/$CRYPTOKI.conf

	dodir /usr/share/bash-completion/completions/
	insinto /usr/share/bash-completion/completions
	doins $S/bash-completion/{$NAME,trust}
}
