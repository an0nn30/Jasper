--- libedit-20251016-1.src/libedit.cygport ---
NAME="libedit"
LICENSE="BSD-3-Clause AND BSD-2-Clause AND ISC"
UPSTREAM_VERS="3.1"
VERSION="20251016"
RELEASE=1
SUMMARY="The NetBSD Editline library"
DESCRIPTION="Libedit is an autotool- and libtoolized port of the NetBSD Editline library.
It provides generic line editing, history, and tokenization functions, similar
to those found in GNU Readline."
HOMEPAGE="https://www.thrysoee.dk/editline/"
SRC_URI="https://www.thrysoee.dk/editline/${NAME}-${VERSION}-${UPSTREAM_VERS}.tar.gz"
SRC_DIR="${NAME}-${VERSION}-${UPSTREAM_VERS}"

PATCH_URI="cygwin-build.patch
	   libedit.patch"

SHLIB_VERSION=0

PKG_NAMES="libedit-devel libedit${SHLIB_VERSION}"

libedit_CATEGORY="Libs"
libedit_REQUIRES="libedit${SHLIB_VERSION}"

libedit_devel_CONTENTS="usr/include usr/lib usr/share/man/man3 usr/share/man/man7"
libedit_devel_CATEGORY="Devel Libs"
libedit_devel_SUMMARY="The NetBSD Editline library (devel)"

# declare needed for dynamic definition of variable name
declare libedit${SHLIB_VERSION}_CONTENTS="usr/bin usr/share/doc usr/share/man/man5"
declare libedit${SHLIB_VERSION}_CATEGORY="Libs"
declare libedit${SHLIB_VERSION}_SUMMARY="The NetBSD Editline library (runtime)"
declare libedit${SHLIB_VERSION}_OBSOLETES="libedit"

WANT_AUTOCONF="2.7"

BUILD_REQUIRES=" \
  autoconf \
  automake \
  gcc-g++ \
  libncurses-devel \
  libtool \
"

SCALLYWAG="notest"
