src_compile() {
    AUTOCONF_VERSION=2.52
#   cd $S
#   inform "updating /usr/bin/libtool to gcc library path $(echo /usr/lib/gcc/$HOSTTYPE-*-$OSTYPE/[0-9]*) if required"
#   chmod a+x $S/update-libtool-gcc.sh	# ensure executable
#   $S/update-libtool-gcc.sh -i		# update libtool in place if required
#   cygautoreconf
    lndirs
    cd $B
    cygconf
    sed -i -e 's%^\(LIBRARIES[ \t]*=\).*$%\1 ../lib/libncursesw.la ../lib/libticw.la%' ncurses/Makefile
    cygmake
}


src_install() {
    cd $B
    cyginstall

    exeinto /usr/lib/ncurses/test
    doexe test/*.exe

    # termcap
#   dodir /etc
#   ./progs/tic -C -t -r ${S}/misc/terminfo.src > ${D}/etc/termcap

    dodir /usr/include/ncurses{,w}

    pushd $D/usr/include/ncurses
    ln -s ../*.h .
    popd

    pushd $D/usr/include/ncursesw
    ln -s ../*.h .
    popd

    pushd $D/usr/lib
    sed -i -e "s| -L[^ '\"]*| |g;s|$D||g" *.la

    for l in lib*a
    do
	ln -s ${l} ${l/w/}
    done

    dosym libncursesw.a		/usr/lib/libcurses.a
    dosym libncursesw.dll.a	/usr/lib/libcurses.dll.a
    popd

    pushd $D/usr/lib/pkgconfig

    for l in *.pc
    do
	ln -s ${l} ${l/w/}
    done

    popd

    pushd $D/usr/bin

    mv ncursesw$CW_NC_ABI-config    ncursesw$NC_ABI-config
    sed -i -e "s@echo \"$CW_NC_ABI\"@echo \"$NC_ABI\"@" ncurses*$NC_ABI-config
    dosym ncursesw$NC_ABI-config    /usr/bin/ncurses$NC_ABI-config
    popd

    pushd $D/usr/share/man/man1
    mv ncursesw$CW_NC_ABI-config.1  ncursesw$NC_ABI-config.1
    dosym ncursesw$NC_ABI-config.1  /usr/share/man/man1/ncurses$NC_ABI-config.1
    popd
}
