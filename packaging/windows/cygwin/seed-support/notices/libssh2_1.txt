#|/usr/bin/cygport
# libssh2.cygport - libssh2 Cygwin package build control script

NAME=libssh2
VERSION=1.11.0
RELEASE=1

CATEGORY="Net"
SUMMARY="SSH2 protocol library"
#XARCH=$ARCH
DESCRIPTION="libssh2 is a library implementing the SSH2 protocol as defined by
Internet Drafts: SECSH-TRANS, SECSH-USERAUTH, SECSH-CONNECTION,
SECSH-ARCH, SECSH-FILEXFER, SECSH-DHGEX, SECSH-NUMBERS, and
SECSH-PUBLICKEY."
#Mingw packages provide MS VC RT-linked binaries, NOT Cygwin binaries,
#for use with the mingw64-{x86_64,i686}-gcc cross compilers, installed in
#/usr/{x86_64,i686}-w64-mingw32/sys-root/mingw/{bin,lib,include}/."

BASE=${NAME##mingw*-*86*-}

HOMEPAGE=https://libssh2.org
# use snapshot date GMT+5 to ensure available
#SNAP=$(TZ=GMT+5 /bin/date +%Y%m%d)
dir=${SNAP:+snapshots}
dir=${dir:-download}
SRC_URI=$HOMEPAGE/$dir/$BASE-$VERSION${SNAP+-$SNAP}.tar.gz
[ -z "$SNAP" ] && SRC_URI+=" $SRC_URI.asc"
GIT_URI=https://github.com/$BASE/$BASE
#inherit git
GIT_ARCHIVE=$GIT_URI/releases/download/$BASE-$VERSION/$BASE-$VERSION.tar.gz
GIT_ARCHIVE+=" $GIT_ARCHIVE.asc"
PATCH_URI="
	libssh2-1.11-configure-ac.patch
	libssh2-1.11-tests-Makefile-am.patch
	libssh2-1.11-tests-Makefile-inc.patch
	libssh2-1.11-tests-CMakeLists-txt.patch
" # tests: Do not run tests accessing internal functions if --disable-static
#	cygwin-configure-ac.patch
#FEDORA=https://src.fedoraproject.org/rpms/$BASE/raw/f34/f
#	$FEDORA/0001-libssh2-1.9.0-CVE-2019-17498.patch

SRC_DIR=$BASE-$VERSION${SNAP+-$SNAP}

CYGCONF_ARGS=--disable-docker-tests

#	--disable-examples-build	Do not build example applications
#	 --enable-fast-install[=PKGS]	optimize for fast installation [default=yes]
#	 --enable-hidden-symbols	Hide internal symbols in library
#	--disable-hidden-symbols	Leave all symbols with default visibility in library
#	--disable-largefile		omit support for large files
#	--disable-libtool-lock		avoid locking (might break parallel builds)
#	 --enable-maintainer-mode	enable make rules and dependencies not useful (and sometimes confusing) to the casual installer
#	--disable-option-checking	ignore unrecognized --enable/--with options
#	 --enable-ossfuzzers		Whether to generate the fuzzers for OSS-Fuzz
#	--disable-rpath			do not hardcode runtime library paths
#	 --enable-shared[=PKGS]		build shared libraries [default=yes]
#	 --enable-silent-rules		less verbose build output (undo: "make V=1")
#	--disable-silent-rules		verbose build output (undo: "make V=0")
#	--disable-sshd-tests		Do not run tests requiring sshd
#	 --enable-static[=PKGS]		build static libraries [default=yes]
#	--disable-tests			Disable tests [default=enabled]
#	--disable-werror		Disable compiler warnings as errors
#	 --enable-werror		Enable compiler warnings as errors
#	   --with-aix-soname=aix|svr4|both	shared library versioning (aka "SONAME") variant to provide on AIX, [default=aix].
#	   --with-crypto=auto|openssl|libgcrypt|mbedtls|wincng|wolfssl	Select crypto backend (default: auto)

#	--without-libssl-prefix		don't search for libssl in includedir and libdir
#	   --with-libssl-prefix[=DIR]	search for libssl in DIR/include and DIR/lib
#	--without-libwolfssl-prefix	don't search for libwolfssl in includedir and libdir
#	   --with-libwolfssl-prefix[=DIR]	search for libwolfssl in DIR/include and DIR/lib

#	--without-libz-prefix		don't search for libz in includedir and libdir
#	   --with-libz-prefix[=DIR]	search for libz in DIR/include and DIR/lib
