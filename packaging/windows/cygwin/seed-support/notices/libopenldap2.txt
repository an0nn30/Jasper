--- openldap-2.6.15-1.src/openldap.cygport ---
# -*- mode: sh; sh-shell: bash -*-
NAME="openldap"
VERSION="2.6.15"
RELEASE="1"
CATEGORY="Net"
SUMMARY="Lightweight Directory Access Protocol suite"
DESCRIPTION="A suite of the Lightweight Directory Access Protocol (v3) servers,
clients, utilities, and development tools."

HOMEPAGE="https://www.openldap.org/"
LICENSE="OLDAP-2.8"
SRC_URI="
  https://www.openldap.org/software/download/OpenLDAP/openldap-release/${P}.tgz
  https://www.openldap.org/software/download/OpenLDAP/openldap-release/${P}.tgz.asc
"
PATCH_URI="
  ${P}-cygwin.patch
"

# Build dependencies only
BUILD_REQUIRES="libcrypt-devel libiodbc-devel libsasl2-devel libssl-devel libuuid-devel"

ABI24="2_4_2"
ABI25="2_5_0"
ABI="2"
PKG_NAMES="${PN} lib${PN}${ABI24} lib${PN}${ABI25} lib${PN}${ABI} ${PN}-devel"
eval ${PN}_SUMMARY='"${SUMMARY} (clients)"'
eval ${PN}_OBSOLETES='"${PN}-server"'
eval ${PN}_CONTENTS='"
  --exclude=devel
  --exclude=drafts
  --exclude=rfc
  usr/bin/ldap*
  usr/share/doc/
  usr/share/man/man1/
"'
eval lib${PN}${ABI}_CATEGORY='"${CATEGORY} Libs"'
eval lib${PN}${ABI}_SUMMARY='"${SUMMARY} (runtime)"'
eval lib${PN}${ABI}_CONTENTS='"usr/bin/*[pr]-${ABI//_/-}.dll"'
eval lib${PN}${ABI24}_CATEGORY='"${CATEGORY} Libs"'
eval lib${PN}${ABI24}_SUMMARY='"${SUMMARY} (runtime)"'
eval lib${PN}${ABI24}_CONTENTS='"usr/bin/*-${ABI24//_/-}.dll"'
eval lib${PN}${ABI25}_CATEGORY='"${CATEGORY} Libs"'
eval lib${PN}${ABI25}_SUMMARY='"${SUMMARY} (runtime)"'
eval lib${PN}${ABI25}_CONTENTS='"usr/bin/*-${ABI25//_/-}.dll"'
eval ${PN}_devel_CATEGORY='"${CATEGORY} Devel Libs"'
eval ${PN}_devel_SUMMARY='"${SUMMARY} (development)"'
eval ${PN}_devel_REQUIRES='"libiodbc-devel openssl-devel libsasl2-devel libuuid-devel"'
eval ${PN}_devel_CONTENTS='"
  usr/include/
  usr/lib/*.a
  usr/lib/pkgconfig/*.pc
  usr/share/doc/${NAME}/[dr]*/
  usr/share/man/man3/
"'

DEPS_PATH="/usr/sbin"
DIFF_EXCLUDES="include tests"
DOCS="ANNOUNCEMENT"

#MAKEOPTS="-j1"

CYGCONF_ARGS="
  --localstatedir=/var/openldap
  --libexecdir=/usr/sbin
  --with-cyrus-sasl
  --enable-dynamic
  --enable-syslog
  --enable-ipv6
  --enable-local
  --disable-cleartext
  --disable-slapd
  --disable-slapi
  --disable-slp
  --disable-mdb
  --disable-relay
  --with-systemd=no
  --disable-wrappers
  --disable-modules
  --disable-overlays
"

src_install() {
    cd ${B}
    cyginstall STRIP=

    dodir /usr/share/doc/${NAME}
    cp -r ${S}/doc/{devel,drafts,guide,rfc} ${D}/usr/share/doc/${NAME}/

    rm -rf ${D}/usr/share/doc/${NAME}/guide
    rm -rf ${D}/etc/openldap
    rm -rf ${D}/usr/share/man/man[58]

    for l in cyg{ldap,lber} ; do
	newbin ${D}/usr/bin/${l}-${ABI}.dll ${l}-${ABI24//_/-}.dll
	[ "cygldap" == "$l" ] && newbin ${D}/usr/bin/${l}-${ABI}.dll ${l}_r-${ABI24//_/-}.dll
	newbin ${D}/usr/bin/${l}-${ABI}.dll ${l}-${ABI25//_/-}.dll
    done

}
