--- cyrus-sasl-2.1.27-1.src/cyrus-sasl.cygport ---
NAME=cyrus-sasl
VERSION=2.1.27
RELEASE=1
DESCRIPTION="sasl2"
HOMEPAGE="http://cyrusimap.org/"

#SRC_URI="ftp://ftp.cyrusimap.org/cyrus-sasl/${P}.tar.gz"
SRC_URI="https://github.com/cyrusimap/cyrus-sasl/releases/download/${P}/${P}.tar.gz"
SRC_URI+=" series libsasl2_3.postinstall"

PATCH_URI="$(grep -v '^#' series)"
#PATCH_URI=

CATEGORY=Utils
SUMMARY="The Cyrus SASL API implementation."
DESCRIPTION="The Cyrus SASL library allows for client/server authentication in conformance with RFC 2222."

PKG_NAMES="cyrus-sasl libsasl2_3 libsasl2_3-sql libsasl2_3-ldap libsasl2-devel"

cyrus_sasl_SUMMARY="${SUMMARY} (Documentation and Utilities)"
cyrus_sasl_CONTENTS='--exclude=saslauthd.exe --exclude=saslauthd.8.gz --exclude=man3 usr/sbin usr/share'

libsasl2_3_SUMMARY="${SUMMARY} (Runtime library and Daemon)"
libsasl2_3_CONTENTS=' --exclude=*ldapdb* --exclude=*sql*
                     etc/sasl2 etc/postinstall usr/bin/*.dll 
                     usr/sbin/saslauthd.exe usr/share/man/man8/saslauthd.8.gz
                     usr/lib/sasl2_3'

libsasl2_devel_SUMMARY="${SUMMARY} (Development files)"
libsasl2_devel_CONTENTS='usr/include usr/lib/libsasl2.dll.a usr/lib/pkgconfig/ usr/share/man/man3'
if defined ARCH_i686; then
  libsasl2_devel_CONTENTS+=' usr/lib/*.la'
fi

libsasl2_3_ldap_SUMMARY="${SUMMARY} (LDAP Plugin)"
libsasl2_3_ldap_CONTENTS='usr/lib/sasl2_3/*ldapdb*'

libsasl2_3_sql_SUMMARY="${SUMMARY} (SQL Plugin)"
libsasl2_3_sql_CONTENTS='usr/lib/sasl2_3/*sql*'

DISTCLEANFILES="config/libtool.m4 config/ltconfig"
DIFF_EXCLUDES="saslauthd.h.in .git"

DEPEND="pkgconfig(openssl) pkgconfig(sqlite3) openldap-devel libpq-devel libdb-devel pkgconfig(krb5-gssapi) libcrypt-devel"

# The configure test for detecting whether dlsym appends a leading
# underscore to the symbol is broken. Override the test by passing
# 'sasl_cv_dlsym_adds_uscore=yes'.
CYGCONF_ARGS="
  --with-configdir=/etc/sasl2
  --with-plugindir=/usr/lib/sasl2_3
  --disable-static --enable-shared
  --enable-sql --with-sqlite3=/usr --with-pgsql=/usr --without-mysql
  --enable-ldapdb --with-ldap=/usr
  --enable-login
  sasl_cv_dlsym_adds_uscore=yes
"

# Build with debugging
if false; then
  inform "Debugging build"
  CFLAGS="-O0 -g"
  CXXFLAGS="-O0 -g"
  RESTRICT=strip debuginfo
fi

#MAKEOPTS+=" -j1"

if defined ARCH_i686; then
  # The ldap .la files reference the sasl2 .la file.
  KEEP_LA_FILES=all
else
  KEEP_LA_FILES=module
fi

ACLOCAL_FLAGS="-I config"

src_install() {
  cd ${B}
  cyginstall

  mv ${D}/usr/lib/sasl2 ${D}/usr/lib/sasl2_3

  dodoc ${S}/saslauthd/LDAP_SASLAUTHD
  dodoc *.html *.txt *.fig

  exeinto /etc/postinstall
  newexe ${S}/libsasl2_3.postinstall libsasl2_3.sh

  # Create an empty /etc/sasl2 as a clue that this is where the
  # config files go.
  dodir /etc/sasl2
  keepdir /etc/sasl2
}

src_test() {
  cd ${B}/utils

  { rm -f sasldb \
    && echo 1234 | ./saslpasswd2 -p -c -f sasldb ken \
    && make testsuite.exe \
    && ./testsuite -a -r 25; } \
  || error "testsuite failed"
}
