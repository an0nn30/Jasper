#|/bin/cygport
# libpsl.cygport - libpsl Cygwin package build control script definitions

NAME=libpsl
VERSION=0.23.3
RELEASE=1

CATEGORY=Net
SUMMARY="Public Suffix List library"
DESCRIPTION="A public suffix is a domain suffix under which Internet users could
or can directly register domain names e.g. .com, .co.uk, and pvt.k12.ma.us.
The Public Suffix List is a list of all known public suffixes."

HOMEPAGE=https://github.com/rockdaboot/libpsl
SRC_URI=$HOMEPAGE/releases/download/$VERSION/$NAME-$VERSION.tar.lz
SRC_URI+=" $SRC_URI.sig"
PATCH_URI="
	makefile-am-install-psl-make-dafsa.patch
	psl-make-dafsa-setlocale.patch
"

# test-is-public-all fails if not all regenerated at build time
DISTCLEANFILES="src/suffixes_dafsa.h tests/psl.dafsa tests/psl_ascii.dafsa"

DIFF_EXCLUDES="tests/psl.dafsa tests/psl_ascii.dafsa"

BUILD_REQUIRES="
	gettext-devel libiconv-devel libicu-devel libidn2-devel libintl-devel
	libunistring-devel libxslt-devel
	gtk-doc publicsuffix-list python3
"

# libpsl is used by curl and wget (which both now use libidn2)
CYGCONF_ARGS="
	 --enable-builtin=libicu
	 --enable-gtk-doc
	 --enable-man
	 --enable-runtime=libidn2
	   --with-psl-distfile=/usr/share/publicsuffix/public_suffix_list.dafsa
	   --with-psl-file=/usr/share/publicsuffix/public_suffix_list.dat
	   --with-psl-testfile=/usr/share/publicsuffix/test_psl.txt
"
#	 --enable-asan		Address Sanitizer (ASan)
#	 --enable-cfi		clang's Control Flow Integrity (CFI)
#	 --enable-fast-install[=PKGS]	optimize for fast installation [default=yes]
#	 --enable-gtk-doc-html	documentation in html format [[default=yes]]
#	 --enable-gtk-doc-pdf	documentation in pdf format [[default=no]]
#	 --enable-runtime=libidn2|libicu|libidn libidn/2=>libunistring/2
#	 --enable-shared[=PKGS]	shared libraries [default=yes]
#	 --enable-static[=PKGS]	static libraries [default=yes]
#	 --enable-ubsan		Undefined Behavior Sanitizer (UBSan)
#	 --enable-valgrind-tests	enable using Valgrind for tests
#	   --with-gnu-ld	assume the C compiler uses GNU ld [default=no]
#	   --with-html-dir=PATH	path to installed docs
#	--without-libiconv-prefix	don't search for libiconv in includedir and libdir
#	   --with-libiconv-prefix[=DIR]	search for libiconv in DIR/include and DIR/lib
#	--without-libintl-prefix	don't search for libintl in includedir and libdir
#	   --with-libintl-prefix[=DIR]	search for libintl in DIR/include and DIR/lib
#	   --with-pic[=PKGS]	try to use only PIC/non-PIC objects [default=use both]
#	   --with-sysroot[=DIR]	dependent libraries in DIR (else compiler's sysroot)

ABI=5

# DAFSA Deterministic Acyclic Finite State Automaton
PKG_NAMES="libpsl-doc libpsl${ABI} libpsl-devel psl psl-make-dafsa"


libpsl_doc_CATEGORY="$CATEGORY Doc"
libpsl_doc_SUMMARY="$SUMMARY (documentation)"
libpsl_doc_CONTENTS="usr/share/gtk-doc/html/libpsl/"


declare libpsl${ABI}_CATEGORY="$CATEGORY Libs"
declare libpsl${ABI}_SUMMARY="$SUMMARY (binaries)"
declare libpsl${ABI}_REQUIRES="publicsuffix-list-dafsa"
declare libpsl${ABI}_CONTENTS="usr/bin/cygpsl-${ABI}.dll usr/share/doc/"


libpsl_devel_CATEGORY="$CATEGORY Libs Devel"
libpsl_devel_SUMMARY="$SUMMARY (development)"
libpsl_devel_CONTENTS="
	usr/include/libpsl.h
	usr/lib/libpsl.dll.a
	usr/lib/pkgconfig/libpsl.pc
	usr/share/man/man3/libpsl.3*
"


psl_CATEGORY="$CATEGORY"
psl_SUMMARY="$SUMMARY (utility)"
psl_OBSOLETES="publicsuffix-utils"
psl_CONTENTS="
	usr/bin/psl.exe
	usr/share/man/man1/psl.1*
"


psl_make_dafsa_CATEGORY="Devel"
psl_make_dafsa_SUMMARY="$SUMMARY optimized DAFSA generator"
psl_make_dafsa_CONTENTS="
	usr/bin/psl-make-dafsa
	usr/share/man/man1/psl-make-dafsa.1*
"
