--- file-5.46-1.src/file.cygport ---
inherit python3

NAME="file"
LICENSE="BSD-2-Clause"
VERSION=5.46
RELEASE=1
CATEGORY="Base"
SUMMARY="Utility for determining file types"
DESCRIPTION="\
The file command is used to identify a particular file according to the
type of data contained by the file.  File can identify many different
file types, including ELF binaries, system libraries, RPM packages, and
different graphics formats."
HOMEPAGE="http://www.darwinsys.com/file/"
SRC_URI="ftp://ftp.astron.com/pub/file/${NAME}-${VERSION}.tar.gz"

PKG_NAMES="${NAME} file-devel python3-magic"
file_CONTENTS="usr/bin/file.exe usr/bin/cygmagic-1.dll usr/share/doc/
	usr/share/file/ usr/share/magic usr/share/man/man[15]/ usr/share/misc/"
file_devel_CATEGORY="Devel"
file_devel_CONTENTS="usr/include/ usr/lib/lib* usr/lib/pkgconfig/libmagic.pc
	usr/share/man/man3/"
python3_magic_OBSOLETES="python2-magic python-magic"
python3_magic_CATEGORY="Python"
python3_magic_REQUIRES="file-devel"  # ctypes.util.find_library('magic')
python3_magic_CONTENTS=${PYTHON3_SITELIB#/}

PATCH_URI_BASE=" https://src.fedoraproject.org/rpms/file/raw/rawhide/f"

PATCH_URI="${PATCH_URI_BASE}/file-4.17-rpm-name.patch"
PATCH_URI+="${PATCH_URI_BASE}/file-5.04-volume_key.patch"
PATCH_URI+="${PATCH_URI_BASE}/file-5.45-readelf-limit-revert.patch"
PATCH_URI+="${PATCH_URI_BASE}/file-5.46-fix-tests-rpm-magic.patch"

CYGCONF_ARGS="--enable-fsect-man5"

BUILD_REQUIRES=" \
  autoconf \
  automake \
  gcc-g++ \
  libbz2-devel \
  liblzma-devel \
  libtool \
  libzstd-devel \
  make \
  python3 \
  zlib-devel \
"
src_install() {
	cd ${B}
	cyginstall

	# install python bindings
	dopython3 ${S}/python/magic.py

	# As on Fedora, create the magic file and the usual links to it.
	set +x
	cat ${S}/magic/Magdir/* > "${D}/usr/share/misc/magic"
	ln -s misc/magic "${D}/usr/share/magic"
	dodir /usr/share/file
	ln -s ../magic "${D}/usr/share/file/magic"
}

SCALLYWAG="notest"
