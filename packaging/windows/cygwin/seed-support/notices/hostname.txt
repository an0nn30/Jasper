--- hostname-3.13-1.cygport ---
# cygport script for hostname

SUMMARY="Show host name, domain name or network addresses"

DESCRIPTION="This package provides the Debian version of the hostname command.
It can be used to display the system's host name, DNS domain name or
network addresses."

CATEGORY="Base"
REQUIRES=""

HOMEPAGE="http://packages.debian.org/unstable/hostname"
SRC_URI="mirror://debian/pool/main/h/hostname/hostname_${PV}.tar.gz"

src_compile() {
	cd ${B}
	lndirs
	cygmake LIBS=
}

src_install() {
	cd ${B}
	dobin hostname.exe
	dosym hostname.exe /usr/bin/domainname
	dosym hostname.exe /usr/bin/dnsdomainname

	doman hostname.1
	dosym hostname.1 /usr/share/man/man1/domainname.1
	dosym hostname.1 /usr/share/man/man1/dnsdomainname.1

	dodoc COPYRIGHT
	docinto debian
	dodoc debian/changelog
}
