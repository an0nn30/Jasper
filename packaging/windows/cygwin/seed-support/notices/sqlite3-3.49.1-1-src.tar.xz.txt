src_compile() {
	cd ${S}
	cygautoreconf
	cd ${S}/tea
	cygautoreconf

	cd ${B}
	cygconf --disable-static-shell --enable-readline --disable-editline --enable-fts5 --enable-json1
	cygmake

	mkdir -p ${B}/tea
	cd ${B}/tea
	CFLAGS+=" -I${S}"
	LIBS+=" -L${B}/.libs"
	cygconf --with-system-sqlite
	cygmake
}

src_install() {
	cd ${B}
	cyginstall
	cd ${B}/tea
	cyginstall
}
