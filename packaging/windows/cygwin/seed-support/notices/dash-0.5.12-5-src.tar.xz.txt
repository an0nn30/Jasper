CYGCONF_ARGS=
#	 --enable-dependency-tracking	do not reject slow dependency extractors
#	--disable-dependency-tracking	speeds up one-time build
#	 --enable-fnmatch		Use fnmatch(3) from libc
#	 --enable-glob			Use glob(3) from libc
#	--disable-lineno		Disable LINENO support
#	--disable-option-checking	ignore unrecognized --enable/--with options
#	 --enable-silent-rules		less verbose build output (undo: "make V=1")
#	--disable-silent-rules		verbose build output (undo: "make V=0")
#	 --enable-static		Build statically linked program
#	 --enable-test-workaround	Guard against faccessat(2) that tells root all files are executable
#	   --with-libedit		Compile with libedit support


# version and date stamp man page
CYGPORT_USE_UNSTABLE_API=1

src_patch_hook()
{
    local ts=`date +'%B %e, %Y'` os=${OSTYPE^}
    local p='s@^\.Dd.*$@.Dd '$ts'@
	    s@^\.Dt.*$@.Dt dash 1 "'$os' User Commands"@
	    s@^\.Os.*$@.Os '$os' "'$NAME' '$PVR'"@'
    inform "applying dynamic patch to version and date stamp man page $os $NAME $PVR $ts"
    echo patching file src/dash.1
    sed -i -e "$p" -- src/dash.1
}


src_install()
{
    cd $B
    cyginstall
    ln $D/usr/bin/dash{,-$PVR}	# add dash-VER-R link as no version info or query
    ln $D/usr/bin/{d,}ash	# link ash name
    ln -s $D/usr/share/man/man1/{d,}ash.1	# ash man symlink to .so in postinstall
}
