cmake_minimum_required(VERSION 3.21)
project(JasperDesktopColors LANGUAGES C CXX)

set(DC_QT_MAJOR "" CACHE STRING "Qt ABI major: 5 or 6")
set(DC_PROBE_KIND "qt" CACHE STRING "Palette provider: qt or kde")
option(DC_BUILD_TESTS "Build native compatibility/tests (never installed)" ON)
set_property(CACHE DC_QT_MAJOR PROPERTY STRINGS 5 6)
set_property(CACHE DC_PROBE_KIND PROPERTY STRINGS qt kde)
if(NOT DC_QT_MAJOR MATCHES "^[56]$")
    message(FATAL_ERROR "DC_QT_MAJOR must be 5 or 6; configure each target in a separate build directory")
endif()
if(NOT DC_PROBE_KIND MATCHES "^(qt|kde)$")
    message(FATAL_ERROR "DC_PROBE_KIND must be qt or kde")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
# Private SDK paths must never become helper runtime search paths.
set(CMAKE_SKIP_RPATH TRUE)
set(CMAKE_FIND_USE_PACKAGE_REGISTRY OFF)
set(CMAKE_FIND_USE_SYSTEM_PACKAGE_REGISTRY OFF)
set(CMAKE_FIND_USE_CMAKE_ENVIRONMENT_PATH OFF)
if(DC_QT_MAJOR STREQUAL "5")
    set(DC_QT_MIN 5.15)
elseif(DC_PROBE_KIND STREQUAL "kde")
    set(DC_QT_MIN 6.5)
else()
    set(DC_QT_MIN 6.2)
endif()
find_package(Qt${DC_QT_MAJOR} ${DC_QT_MIN} QUIET CONFIG COMPONENTS Widgets)
if(NOT TARGET Qt${DC_QT_MAJOR}::Widgets)
    message(FATAL_ERROR "Missing desktop-colors SDK: Qt${DC_QT_MAJOR} Widgets >= ${DC_QT_MIN}. Install development packages or set the target's jasper.*Prefix; no SDK is downloaded automatically.")
endif()

set(DC_LIBRARIES Qt${DC_QT_MAJOR}::Widgets)
if(DC_PROBE_KIND STREQUAL "kde")
    if(DC_QT_MAJOR STREQUAL "5")
        find_package(KF5Config 5.92 QUIET CONFIG)
        find_package(KF5ConfigWidgets 5.92 QUIET CONFIG)
        if(NOT TARGET KF5::ConfigCore OR NOT TARGET KF5::ConfigWidgets)
            message(FATAL_ERROR "Missing desktop-colors SDK: KF5 ConfigWidgets/Config >= 5.92; set jasper.kf5Prefix.")
        endif()
        list(APPEND DC_LIBRARIES KF5::ConfigCore KF5::ConfigWidgets)
    else()
        find_package(KF6Config 6.0 QUIET CONFIG)
        find_package(KF6ColorScheme 6.0 QUIET CONFIG)
        if(NOT TARGET KF6::ConfigCore OR NOT TARGET KF6::ColorScheme)
            message(FATAL_ERROR "Missing desktop-colors SDK: KF6 ColorScheme/Config >= 6.0; set jasper.kf6Prefix and jasper.kde6QtPrefix.")
        endif()
        list(APPEND DC_LIBRARIES KF6::ConfigCore KF6::ColorScheme)
    endif()
    include(CheckCXXSourceCompiles)
    set(CMAKE_REQUIRED_LIBRARIES ${DC_LIBRARIES})
    # Test the public enum AND constructor, rather than guessing from a version number.
    check_cxx_source_compiles("#include <KColorScheme>
        int main() { KColorScheme s(QPalette::Active, KColorScheme::Header); return 0; }"
        DC_HAVE_KCOLORSCHEME_HEADER)
    unset(CMAKE_REQUIRED_LIBRARIES)
    message(STATUS "KColorScheme Header public API: ${DC_HAVE_KCOLORSCHEME_HEADER}")
endif()

if(DC_BUILD_TESTS)
    enable_testing()
    add_executable(qt-compatibility test/qt_compatibility.cpp)
    target_link_libraries(qt-compatibility PRIVATE ${DC_LIBRARIES})
    if(DC_PROBE_KIND STREQUAL "kde")
        target_compile_definitions(qt-compatibility PRIVATE DC_KDE=1
            DC_HAVE_KCOLORSCHEME_HEADER=$<BOOL:${DC_HAVE_KCOLORSCHEME_HEADER}>)
    endif()
    add_test(NAME desktop-qt-compatibility COMMAND qt-compatibility)
    set_tests_properties(desktop-qt-compatibility PROPERTIES TIMEOUT 20)
endif()

if(DC_PROBE_KIND STREQUAL "qt")
    add_library(desktop-qt STATIC src/qt_probe.cpp src/protocol.c)
    target_include_directories(desktop-qt PUBLIC src)
    target_link_libraries(desktop-qt PUBLIC Qt${DC_QT_MAJOR}::Widgets)
    target_compile_options(desktop-qt PRIVATE -Wall -Wextra -Werror)
    add_executable(jasper-desktop-qt${DC_QT_MAJOR} src/qt_main.cpp)
    target_link_libraries(jasper-desktop-qt${DC_QT_MAJOR} PRIVATE desktop-qt)
    set_target_properties(jasper-desktop-qt${DC_QT_MAJOR} PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/production")
    if(DC_BUILD_TESTS)
        find_package(Python3 REQUIRED COMPONENTS Interpreter)
        add_executable(qt-probe-test test/qt_probe_test.cpp)
        target_link_libraries(qt-probe-test PRIVATE desktop-qt)
        add_executable(qt-fixture test/qt-fixtures/qt_fixture_main.cpp)
        target_link_libraries(qt-fixture PRIVATE desktop-qt)
        add_test(NAME desktop-qt-extraction COMMAND qt-probe-test)
        add_test(NAME desktop-qt-process COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/test/qt_process_test.py
            $<TARGET_FILE:jasper-desktop-qt${DC_QT_MAJOR}> $<TARGET_FILE:qt-fixture> qt${DC_QT_MAJOR})
        set_tests_properties(desktop-qt-extraction desktop-qt-process PROPERTIES TIMEOUT 60)
    endif()
endif()
